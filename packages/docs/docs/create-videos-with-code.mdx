import React from "react";
import {interpolate, spring, useCurrentFrame, useVideoConfig} from "remotion";

const CHUNK = 3; // palabras visibles a la vez

export const Subtitles: React.FC<{text: string}> = ({text}) => {
  const frame = useCurrentFrame();
  const {durationInFrames, fps} = useVideoConfig();
  const words = text.split(" ");

  // Reparto uniforme de palabras. Si usas voz real, sustituye por tiempos exactos.
  const startAt = 6;
  const endAt = durationInFrames * 0.92;
  const progress = Math.min(1, Math.max(0, (frame - startAt) / (endAt - startAt)));
  const active = Math.min(words.length - 1, Math.floor(progress * words.length));
  const chunk = Math.floor(active / CHUNK);
  const visible = words.slice(chunk * CHUNK, chunk * CHUNK + CHUNK);

  const wordStart = startAt + (active / words.length) * (endAt - startAt);
  const pop = spring({frame: frame - wordStart, fps, config: {damping: 12, stiffness: 200}});
  const popScale = interpolate(pop, [0, 1], [1.3, 1]);

  return (
    <div
      style={{
        position: "absolute",
        left: 60,
        right: 60,
        bottom: 420, // zona segura: no tapar la interfaz de TikTok/Reels/Shorts
        display: "flex",
        justifyContent: "center",
        flexWrap: "wrap",
        gap: 20,
      }}
    >
      {visible.map((w, i) => {
        const idx = chunk * CHUNK + i;
        const isActive = idx === active;
        return (
          <span
            key={idx}
            style={{
              opacity: idx <= active ? 1 : 0,
              color: isActive ? "#FFD400" : "#FFFFFF",
              transform: `scale(${isActive ? popScale : 1})`,
              fontFamily: "Impact, 'Arial Black', sans-serif",
              fontSize: 100,
              fontWeight: 900,
              textTransform: "uppercase",
              WebkitTextStroke: "8px #000",
              paintOrder: "stroke fill",
              textShadow: "0 10px 28px rgba(0,0,0,0.75)",
              lineHeight: 1.05,
            }}
          >
            {w}
          </span>
        );
      })}
    </div>
  );
};

import React from "react";
import {
  AbsoluteFill,
  Audio,
  Easing,
  OffthreadVideo,
  Sequence,
  interpolate,
  staticFile,
  useCurrentFrame,
  useVideoConfig,
} from "remotion";
import {FPS, HAS_MUSIC, HAS_VOICE, SEGS, STARTS, TOTAL, peakFrame, segFrames, Seg} from "./data";
import {Subtitles} from "./Subtitles";

const Segment: React.FC<{seg: Seg}> = ({seg}) => {
  const frame = useCurrentFrame();
  const {durationInFrames} = useVideoConfig();
  const t = frame / durationInFrames;
  const peak = peakFrame(seg);
  const since = frame - peak;

  // Zoom lento + golpe de zoom en el impacto
  const zoom = interpolate(t, [0, 1], seg.zoom, {easing: Easing.out(Easing.quad)});
  const punch = since >= 0 ? 0.08 * Math.max(0, 1 - since / 10) : 0;

  // Temblor de cámara que se apaga tras el impacto
  const decay = since >= 0 ? Math.max(0, 1 - since / 20) : 0;
  const dx = Math.sin(frame * 1.9) * seg.shake * decay;
  const dy = Math.cos(frame * 2.7) * seg.shake * decay;

  // Destello blanco en el impacto
  const flash = seg.flash
    ? interpolate(frame, [peak, peak + 2, peak + 10], [0, 0.85, 0], {
        extrapolateLeft: "clamp",
        extrapolateRight: "clamp",
      })
    : 0;

  // Título solo en el primer corte
  const titleOpacity =
    seg.id === 1
      ? interpolate(frame, [0, 10, 40, 55], [0, 1, 1, 0], {extrapolateRight: "clamp"})
      : 0;

  return (
    <AbsoluteFill style={{backgroundColor: "#000"}}>
      <AbsoluteFill style={{transform: `translate(${dx}px, ${dy}px) scale(${zoom + punch})`}}>
        <OffthreadVideo
          src={staticFile("clip.mp4")}
          startFrom={Math.round(seg.src * FPS)}
          playbackRate={seg.rate}
          volume={0.5}
          style={{
            width: "100%",
            height: "100%",
            objectFit: "cover", // 2340x1080 -> 1080x1920, recorte centrado
            objectPosition: `${seg.focusX}% 50%`,
            filter: "contrast(1.18) saturate(1.25) brightness(0.95)",
          }}
        />
      </AbsoluteFill>

      {/* Viñeta oscura para look cinematográfico */}
      <AbsoluteFill
        style={{
          background: "radial-gradient(ellipse at center, rgba(0,0,0,0) 45%, rgba(0,0,0,0.65) 100%)",
        }}
      />

      <AbsoluteFill style={{backgroundColor: "#fff", opacity: flash}} />

      {seg.id === 1 && (
        <AbsoluteFill style={{justifyContent: "flex-start", alignItems: "center", paddingTop: 260, opacity: titleOpacity}}>
          <div
            style={{
              fontFamily: "Impact, 'Arial Black', sans-serif",
              fontSize: 130,
              color: "#FFD400",
              WebkitTextStroke: "8px #000",
              paintOrder: "stroke fill",
              textAlign: "center",
              lineHeight: 1,
            }}
          >
            EL ÚLTIMO
            <br />
            DISPARO
          </div>
        </AbsoluteFill>
      )}

      <Subtitles text={seg.text} />

      {HAS_VOICE && <Audio src={staticFile(`voz/${String(seg.id).padStart(2, "0")}.mp3`)} />}
    </AbsoluteFill>
  );
};

export const Epico: React.FC = () => (
  <AbsoluteFill style={{backgroundColor: "#000"}}>
    {SEGS.map((seg, i) => (
      <Sequence key={seg.id} from={STARTS[i]} durationInFrames={segFrames(seg)}>
        <Segment seg={seg} />
      </Sequence>
    ))}
    {HAS_MUSIC && (
      <Audio
        src={staticFile("musica.mp3")}
        volume={(f) =>
          interpolate(f, [0, TOTAL * 0.7, TOTAL - 30, TOTAL], [0.2, 0.5, 0.7, 0], {
            extrapolateLeft: "clamp",
            extrapolateRight: "clamp",
          })
        }
      />
    )}
  </AbsoluteFill>
);
EOF

// Revisión fotograma a fotograma: exporta stills de cada corte en out/review/
import {execSync} from "node:child_process";
import {mkdirSync} from "node:fs";
import {SEGS, STARTS, segFrames, peakFrame} from "../src/data";

mkdirSync("out/review", {recursive: true});

SEGS.forEach((s, i) => {
  const a = STARTS[i];
  const n = segFrames(s);
  const p = a + peakFrame(s);
  const frames: Record<string, number> = {
    "1_inicio": a + 1,
    "2_pico": p,
    "3_pico_mas5": p + 5,
    "4_final": a + n - 2,
  };
  for (const [name, f] of Object.entries(frames)) {
    const id = String(s.id).padStart(2, "0");
    execSync(
      `npx remotion still src/index.ts Epico out/review/seg${id}_${name}.png --frame=${f}`,
      {stdio: "inherit"}
    );
  }
});
console.log("Listo: revisa la carpeta out/review/");
EOF

# Videos IA — "El último disparo" (vertical 1080x1920, 30 fps, ~40 s)

## 1. Preparar el clip (obligatorio)
El original es de fps variable (VFR) y Remotion lo necesita constante:

    ffmpeg -i TU_CLIP.mp4 -vf "scale=-2:1080" -r 30 -c:v libx264 -crf 20 -c:a aac -ar 48000 public/clip.mp4

## 2. Instalar y previsualizar
    npm install
    npm run studio

## 3. Voz en español
Genera un audio por corte con el texto de src/data.ts (campo `text`) y guárdalos como
public/voz/01.mp3 ... 10.mp3. Luego pon HAS_VOICE = true en src/data.ts.
Tono: grave, pausado, con subida de intensidad hacia los cortes 7 y 10.
Música épica opcional en public/musica.mp3 (HAS_MUSIC = true).

## 4. Revisión de cada fotograma
    npm run review
Crea out/review/segXX_*.png (inicio, impacto, impacto+5, final de cada corte).
Revisa sobre todo los cortes 7 y 10 (cámara lenta): si el clip no empieza donde
debe, ajusta `src` en src/data.ts.

## 5. Render final
    npm run render      ->  out/epico.mp4
